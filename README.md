backend Setup:
1. cd backend
2. python -m venv venv
3. venv/scripts/activate
4. pip install -r requirements.txt 
5. python run.py
6. pip install google-genai
7. pip install python-dotenv


frontend Setup:  
1. npm run dev
2. npm install three @react-three/fiber @react-three/drei
3. npm install framer-motion
4. npm install jspdf html2canvas
5. npm install react-leaflet leaflet
6. npm install lucide-react
7. npm install country-state-city
